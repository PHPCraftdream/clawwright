_clawwright_complete() {
  local cur="${COMP_WORDS[COMP_CWORD]}"
  local prev="${COMP_WORDS[COMP_CWORD-1]}"
  local words=("${COMP_WORDS[@]}")
  local cword=$COMP_CWORD
  local cmd="" idx=0 skip=0
  local between=() scope="" reply=""
  for ((i = 1; i < cword; i++)); do
    local w="${words[i]}"
    if [[ $skip -eq 1 ]]; then skip=0; continue; fi
    if [[ "$w" == "--app" ]]; then skip=1; continue; fi
    if [[ "$w" == --app=* ]]; then continue; fi
    cmd="$w"; idx=$i; break
  done
  if [[ -z "$cmd" ]]; then
    if [[ "$prev" == "--app" ]]; then
    COMPREPLY=( $(compgen -W "$("${COMP_WORDS[0]}" list --json --no-status 2>/dev/null | grep -o '"name":"[^"]*"' | cut -d'"' -f4 | grep -v '^[.]')" -- "$cur") )
    return
  fi
    if _clawwright_lookup "top"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  if (( cword - idx - 1 > 0 )); then between=("${words[@]:$((idx + 1)):$((cword - idx - 1))}"); fi
  # The command's own first candidates; a command the table does not carry completes to
  # nothing at all.
  if _clawwright_lookup "first $cmd"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); return; fi
  # Nothing after the command: the word being completed IS its own first position, which
  # the lookup above already answered.
  if (( ${#between[@]} == 0 )); then return; fi
  # The words typed so far, read the way the parser reads them: an option takes the next word
  # as its value (so a `--` there is no marker), a bare `--` ends the options, and a
  # pass-through command's literal tail starts at its first undeclared dash word or its first
  # word past the declared positionals. Missing arms answer empty.
  local opts="" flags="" verb="" isverb=0 ended=0 swallow=0 tail=0 free=0 token key inline
  if _clawwright_lookup "valueOptions $cmd"; then opts="$_clawwright_reply"; fi
  if _clawwright_lookup "verbatim $cmd"; then isverb=1; verb="$_clawwright_reply"; fi
  if _clawwright_lookup "verbatimFlags $cmd"; then flags="$_clawwright_reply"; fi
  for token in "${between[@]}"; do
    if (( swallow )); then swallow=0; continue; fi
    if [[ "$token" == "--" ]]; then ended=1; break; fi
    key="$token"; inline=0
    if [[ "$token" == --*=* ]]; then key="${token%%=*}"; inline=1; fi
    if (( ! inline )) && [[ -n "$token" && " $opts " == *" $token "* ]]; then swallow=1; continue; fi
    if [[ "$token" == -* ]]; then
      if (( isverb )) && [[ " $flags " != *" $key "* ]]; then tail=1; break; fi
      continue
    fi
    free=$((free + 1))
    if (( isverb && free > verb )); then tail=1; break; fi
  done
  if (( ended || tail )); then COMPREPLY=(); return; fi
  # The first word names the action; an option still waiting for its value completes that
  # value (keyed command + action + option), or nothing when it has no choices.
  if [[ "${between[0]}" != -* ]] && _clawwright_lookup "after $cmd ${between[0]}"; then scope="${between[0]}"; fi
  if (( swallow )); then
    if _clawwright_lookup "values ${cmd}${scope}${prev}"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  # Past a word after the command: that word's own list, else the command's fallback.
  if _clawwright_lookup "after $cmd ${between[0]}"; then
    COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") )
  elif _clawwright_lookup "after $cmd *"; then
    COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") )
  else
    COMPREPLY=()
  fi
}

_clawwright_complete() {
  local cur="${COMP_WORDS[COMP_CWORD]}"
  local prev="${COMP_WORDS[COMP_CWORD-1]}"
  local words=("${COMP_WORDS[@]}")
  local cword=$COMP_CWORD
  local cmd="" idx=0 skip=0
  local between=() scope="" reply=""
  for ((i = 1; i < cword; i++)); do
    local w="${words[i]}"
    if [[ $skip -eq 1 ]]; then skip=0; continue; fi
    if [[ "$w" == "--app" ]]; then skip=1; continue; fi
    if [[ "$w" == --app=* ]]; then continue; fi
    cmd="$w"; idx=$i; break
  done
  if [[ -z "$cmd" ]]; then
    if [[ "$prev" == "--app" ]]; then
    COMPREPLY=( $(compgen -W "$("${COMP_WORDS[0]}" list --json --no-status 2>/dev/null | grep -o '"name":"[^"]*"' | cut -d'"' -f4 | grep -v '^[.]')" -- "$cur") )
    return
  fi
    if _clawwright_lookup "top"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  if (( cword - idx - 1 > 0 )); then between=("${words[@]:$((idx + 1)):$((cword - idx - 1))}"); fi
  # The command's own first candidates; a command the table does not carry completes to
  # nothing at all.
  if _clawwright_lookup "first $cmd"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); return; fi
  # Nothing after the command: the word being completed IS its own first position, which
  # the lookup above already answered.
  if (( ${#between[@]} == 0 )); then return; fi
  # The words typed so far, read the way the parser reads them: an option takes the next word
  # as its value (so a `--` there is no marker), a bare `--` ends the options, and a
  # pass-through command's literal tail starts at its first undeclared dash word or its first
  # word past the declared positionals. Missing arms answer empty.
  local opts="" flags="" verb="" isverb=0 ended=0 swallow=0 tail=0 free=0 token key inline
  if _clawwright_lookup "valueOptions $cmd"; then opts="$_clawwright_reply"; fi
  if _clawwright_lookup "verbatim $cmd"; then isverb=1; verb="$_clawwright_reply"; fi
  if _clawwright_lookup "verbatimFlags $cmd"; then flags="$_clawwright_reply"; fi
  for token in "${between[@]}"; do
    if (( swallow )); then swallow=0; continue; fi
    if [[ "$token" == "--" ]]; then ended=1; break; fi
    key="$token"; inline=0
    if [[ "$token" == --*=* ]]; then key="${token%%=*}"; inline=1; fi
    if (( ! inline )) && [[ -n "$token" && " $opts " == *" $token "* ]]; then swallow=1; continue; fi
    if [[ "$token" == -* ]]; then
      if (( isverb )) && [[ " $flags " != *" $key "* ]]; then tail=1; break; fi
      continue
    fi
    free=$((free + 1))
    if (( isverb && free > verb )); then tail=1; break; fi
  done
  if (( ended || tail )); then COMPREPLY=(); return; fi
  # The first word names the action; an option still waiting for its value completes that
  # value (keyed command + action + option), or nothing when it has no choices.
  if [[ "${between[0]}" != -* ]] && _clawwright_lookup "after $cmd ${between[0]}"; then scope="${between[0]}"; fi
  if (( swallow )); then
    if _clawwright_lookup "values ${cmd}${scope}${prev}"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  # Past a word after the command: that word's own list, else the command's fallback.
  if _clawwright_lookup "after $cmd ${between[0]}"; then
    COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") )
  elif _clawwright_lookup "after $cmd *"; then
    COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") )
  else
    COMPREPLY=()
  fi
}
complete -F _clawwright_complete clawwright
complete -F _clawwright_complete ./clawwright

$clawwrightCompleter = {
  param($wordToComplete, $commandAst, $cursorPosition)
  $tokens = @($commandAst.CommandElements | ForEach-Object { $_.Extent.Text })
  $rest = @(if ($tokens.Count -gt 1) { @($tokens[1..($tokens.Count - 1)]) } else { @() })
  # Only the words BEFORE the word being completed pick the command — the partial word
  # itself is not a typed command (R32-02: 'clawwright sta<Tab>' saw the command "sta").
  $scan = $rest
  if ($wordToComplete -ne '') {
    $scan = @(if ($rest.Count -gt 1) { @($rest[0..($rest.Count - 2)]) } else { @() })
  }
  $prev = $null
  if ($scan.Count -ge 1) { $prev = $scan[$scan.Count - 1] }
  # --app is positional and takes a value: the pair is skipped, so it is never read as the
  # command (and a value that looks like a command name does not shadow the real one).
  # The '--app=<name>' form carries its own value in one token: it is skipped alone.
  $i = 0
  while ($clawwrightApp -and $i -lt $scan.Count -and ($scan[$i] -ceq '--app' -or $scan[$i].StartsWith('--app='))) { $i = $i + $(if ($scan[$i] -ceq '--app') { 2 } else { 1 }) }
  $candidates = @()
  if ($i -ge $scan.Count) {
    if ($clawwrightApp -and $prev -ceq '--app') {
      # --app's own value, lazily: only while no command word has been typed — --app must
      # come before the command, so past one the command's flags return (R33-10). Whichever
      # of the system-wide command or the checkout shim was typed makes the call.
      $names = @(try { & $tokens[0] list --json --no-status 2>$null | ConvertFrom-Json | ForEach-Object { $_.name } | Where-Object { $_ -notlike '.*' } } catch { @() })
      $names | Where-Object { $_ -clike "$wordToComplete*" } | ForEach-Object {
        [System.Management.Automation.CompletionResult]::new($_, $_, 'ParameterValue', $_)
      }
      return
    }
    $candidates = @($clawwrightTop)
  } else {
    $cmd = $scan[$i]
    $between = @()
    if ($scan.Count - $i - 1 -gt 0) { $between = @($scan[($i + 1)..($scan.Count - 1)]) }
    # The words typed so far, read the way the parser reads them: an option takes the next
    # word as its value (so a `--` there is no marker), a bare `--` ends the options, and a
    # pass-through command's literal tail starts at its first undeclared dash word or its
    # first word past the declared positionals.
    $ended = $false
    $swallow = $false
    $tail = $false
    $free = 0
    $isVerbatim = $clawwrightVerbatim.ContainsKey($cmd)
    foreach ($w2 in $between) {
      if ($swallow) { $swallow = $false; continue }
      if ($w2 -ceq '--') { $ended = $true; break }
      $key = $w2
      $inline = $false
      if ($w2.StartsWith('--') -and $w2.Contains('=')) { $key = $w2.Substring(0, $w2.IndexOf('=')); $inline = $true }
      if ((-not $inline) -and ($w2 -cne '') -and ($clawwrightValueOptions[$cmd] -ccontains $w2)) { $swallow = $true; continue }
      if ($w2.StartsWith('-')) {
        if ($isVerbatim -and -not ($clawwrightVerbatimFlags[$cmd] -ccontains $key)) { $tail = $true; break }
        continue
      }
      $free = $free + 1
      if ($isVerbatim -and $free -gt $clawwrightVerbatim[$cmd]) { $tail = $true; break }
    }
    # The first word names the action; an option still waiting for its value completes that
    # value (keyed command + action + option), or nothing when it has no choices.
    $scope = ''
    $afterKey = "$cmd $($between[0])"
    if ($between.Count -gt 0 -and $clawwrightAfter.ContainsKey($afterKey)) { $scope = $between[0] }
    if ($between.Count -eq 0) {
      # An unknown command reaches `first` with nothing under that key: the @() keeps the
      # type an array and the filter below drops the $null — nothing is offered.
      $candidates = @($clawwrightFirst[$cmd])
    } elseif ($ended -or $tail) {
      $candidates = @()
    } elseif ($swallow) {
      $key = "$cmd$scope$prev"
      if ($clawwrightValues.ContainsKey($key)) { $candidates = @($clawwrightValues[$key]) } else { $candidates = @() }
    } elseif ($clawwrightAfter.ContainsKey($afterKey)) {
      $candidates = @($clawwrightAfter[$afterKey])
    } else {
      $candidates = @($clawwrightAfter["$cmd *"])
    }
  }
  $candidates | Where-Object { $_ -clike "$wordToComplete*" } | Sort-Object -Unique -CaseSensitive | ForEach-Object {
    [System.Management.Automation.CompletionResult]::new($_, $_, 'ParameterValue', $_)
  }
}
Register-ArgumentCompleter -Native -CommandName clawwright, ./clawwright -ScriptBlock $clawwrightCompleter

_clawwright_complete() {
  local cur="${COMP_WORDS[COMP_CWORD]}"
  local prev="${COMP_WORDS[COMP_CWORD-1]}"
  local words=("${COMP_WORDS[@]}")
  local cword=$COMP_CWORD
  local cmd="" idx=0 skip=0
  local between=() scope="" reply=""
  for ((i = 1; i < cword; i++)); do
    local w="${words[i]}"
        cmd="$w"; idx=$i; break
  done
  if [[ -z "$cmd" ]]; then
        if _clawwright_lookup "top"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  if (( cword - idx - 1 > 0 )); then between=("${words[@]:$((idx + 1)):$((cword - idx - 1))}"); fi
  # The command's own first candidates; a command the table does not carry completes to
  # nothing at all.
  if _clawwright_lookup "first $cmd"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); return; fi
  # Nothing after the command: the word being completed IS its own first position, which
  # the lookup above already answered.
  if (( ${#between[@]} == 0 )); then return; fi
  # The words typed so far, read the way the parser reads them: an option takes the next word
  # as its value (so a `--` there is no marker), a bare `--` ends the options, and a
  # pass-through command's literal tail starts at its first undeclared dash word or its first
  # word past the declared positionals. Missing arms answer empty.
  local opts="" flags="" verb="" isverb=0 ended=0 swallow=0 tail=0 free=0 token key inline
  if _clawwright_lookup "valueOptions $cmd"; then opts="$_clawwright_reply"; fi
  if _clawwright_lookup "verbatim $cmd"; then isverb=1; verb="$_clawwright_reply"; fi
  if _clawwright_lookup "verbatimFlags $cmd"; then flags="$_clawwright_reply"; fi
  for token in "${between[@]}"; do
    if (( swallow )); then swallow=0; continue; fi
    if [[ "$token" == "--" ]]; then ended=1; break; fi
    key="$token"; inline=0
    if [[ "$token" == --*=* ]]; then key="${token%%=*}"; inline=1; fi
    if (( ! inline )) && [[ -n "$token" && " $opts " == *" $token "* ]]; then swallow=1; continue; fi
    if [[ "$token" == -* ]]; then
      if (( isverb )) && [[ " $flags " != *" $key "* ]]; then tail=1; break; fi
      continue
    fi
    free=$((free + 1))
    if (( isverb && free > verb )); then tail=1; break; fi
  done
  if (( ended || tail )); then COMPREPLY=(); return; fi
  # The first word names the action; an option still waiting for its value completes that
  # value (keyed command + action + option), or nothing when it has no choices.
  if [[ "${between[0]}" != -* ]] && _clawwright_lookup "after $cmd ${between[0]}"; then scope="${between[0]}"; fi
  if (( swallow )); then
    if _clawwright_lookup "values ${cmd}${scope}${prev}"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  # Past a word after the command: that word's own list, else the command's fallback.
  if _clawwright_lookup "after $cmd ${between[0]}"; then
    COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") )
  elif _clawwright_lookup "after $cmd *"; then
    COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") )
  else
    COMPREPLY=()
  fi
}
complete -F _clawwright_complete clawwright
complete -F _clawwright_complete ./clawwright

_clawwright_complete() {
  local cur="${COMP_WORDS[COMP_CWORD]}"
  local prev="${COMP_WORDS[COMP_CWORD-1]}"
  local words=("${COMP_WORDS[@]}")
  local cword=$COMP_CWORD
  local cmd="" idx=0 skip=0
  local between=() scope="" reply=""
  for ((i = 1; i < cword; i++)); do
    local w="${words[i]}"
        cmd="$w"; idx=$i; break
  done
  if [[ -z "$cmd" ]]; then
        if _clawwright_lookup "top"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  if (( cword - idx - 1 > 0 )); then between=("${words[@]:$((idx + 1)):$((cword - idx - 1))}"); fi
  # The command's own first candidates; a command the table does not carry completes to
  # nothing at all.
  if _clawwright_lookup "first $cmd"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); return; fi
  # Nothing after the command: the word being completed IS its own first position, which
  # the lookup above already answered.
  if (( ${#between[@]} == 0 )); then return; fi
  # The words typed so far, read the way the parser reads them: an option takes the next word
  # as its value (so a `--` there is no marker), a bare `--` ends the options, and a
  # pass-through command's literal tail starts at its first undeclared dash word or its first
  # word past the declared positionals. Missing arms answer empty.
  local opts="" flags="" verb="" isverb=0 ended=0 swallow=0 tail=0 free=0 token key inline
  if _clawwright_lookup "valueOptions $cmd"; then opts="$_clawwright_reply"; fi
  if _clawwright_lookup "verbatim $cmd"; then isverb=1; verb="$_clawwright_reply"; fi
  if _clawwright_lookup "verbatimFlags $cmd"; then flags="$_clawwright_reply"; fi
  for token in "${between[@]}"; do
    if (( swallow )); then swallow=0; continue; fi
    if [[ "$token" == "--" ]]; then ended=1; break; fi
    key="$token"; inline=0
    if [[ "$token" == --*=* ]]; then key="${token%%=*}"; inline=1; fi
    if (( ! inline )) && [[ -n "$token" && " $opts " == *" $token "* ]]; then swallow=1; continue; fi
    if [[ "$token" == -* ]]; then
      if (( isverb )) && [[ " $flags " != *" $key "* ]]; then tail=1; break; fi
      continue
    fi
    free=$((free + 1))
    if (( isverb && free > verb )); then tail=1; break; fi
  done
  if (( ended || tail )); then COMPREPLY=(); return; fi
  # The first word names the action; an option still waiting for its value completes that
  # value (keyed command + action + option), or nothing when it has no choices.
  if [[ "${between[0]}" != -* ]] && _clawwright_lookup "after $cmd ${between[0]}"; then scope="${between[0]}"; fi
  if (( swallow )); then
    if _clawwright_lookup "values ${cmd}${scope}${prev}"; then COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") ); else COMPREPLY=(); fi
    return
  fi
  # Past a word after the command: that word's own list, else the command's fallback.
  if _clawwright_lookup "after $cmd ${between[0]}"; then
    COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") )
  elif _clawwright_lookup "after $cmd *"; then
    COMPREPLY=( $(compgen -W "$_clawwright_reply" -- "$cur") )
  else
    COMPREPLY=()
  fi
}
complete -F _clawwright_complete clawwright
complete -F _clawwright_complete ./clawwright

  "set" = @("--name", "--set", "--from", "--to", "--set-id", "--receipt", "--kind", "--break-foreign-lock")
}, [System.StringComparer]::Ordinal)
$clawwrightValues = [hashtable]::new(@{
  "backup--profile" = @("full", "migrate", "share")
  "backupcreate--profile" = @("full", "migrate", "share")
  "pull--profile" = @("full", "migrate", "share")
  "verify--profile" = @("full", "migrate", "share")
  "memoryplan--mode" = @("embeddings", "keyword", "disabled")
  "memoryconfigure--mode" = @("embeddings", "keyword", "disabled")
  "memoryplan--backend" = @("builtin")
  "memoryconfigure--backend" = @("builtin")
  "mcp-setup--client" = @("claude", "codex", "both")
  "setforget--kind" = @("agent", "mcp-server", "cron-job")
}, [System.StringComparer]::Ordinal)
$clawwrightCompleter = {
  param($wordToComplete, $commandAst, $cursorPosition)
  $tokens = @($commandAst.CommandElements | ForEach-Object { $_.Extent.Text })
  $rest = @(if ($tokens.Count -gt 1) { @($tokens[1..($tokens.Count - 1)]) } else { @() })
  # Only the words BEFORE the word being completed pick the command — the partial word
  # itself is not a typed command (R32-02: 'clawwright sta<Tab>' saw the command "sta").
  $scan = $rest
  if ($wordToComplete -ne '') {
    $scan = @(if ($rest.Count -gt 1) { @($rest[0..($rest.Count - 2)]) } else { @() })
  }
  $prev = $null
  if ($scan.Count -ge 1) { $prev = $scan[$scan.Count - 1] }
  # --app is positional and takes a value: the pair is skipped, so it is never read as the
  # command (and a value that looks like a command name does not shadow the real one).
  # The '--app=<name>' form carries its own value in one token: it is skipped alone.
  $i = 0
  while ($clawwrightApp -and $i -lt $scan.Count -and ($scan[$i] -ceq '--app' -or $scan[$i].StartsWith('--app='))) { $i = $i + $(if ($scan[$i] -ceq '--app') { 2 } else { 1 }) }
  $candidates = @()
  if ($i -ge $scan.Count) {
    if ($clawwrightApp -and $prev -ceq '--app') {
      # --app's own value, lazily: only while no command word has been typed — --app must
      # come before the command, so past one the command's flags return (R33-10). Whichever
      # of the system-wide command or the checkout shim was typed makes the call.
      $names = @(try { & $tokens[0] list --json --no-status 2>$null | ConvertFrom-Json | ForEach-Object { $_.name } | Where-Object { $_ -notlike '.*' } } catch { @() })
      $names | Where-Object { $_ -clike "$wordToComplete*" } | ForEach-Object {
        [System.Management.Automation.CompletionResult]::new($_, $_, 'ParameterValue', $_)
      }
      return
    }
    $candidates = @($clawwrightTop)
  } else {
    $cmd = $scan[$i]
    $between = @()
    if ($scan.Count - $i - 1 -gt 0) { $between = @($scan[($i + 1)..($scan.Count - 1)]) }
    # The words typed so far, read the way the parser reads them: an option takes the next
    # word as its value (so a `--` there is no marker), a bare `--` ends the options, and a
    # pass-through command's literal tail starts at its first undeclared dash word or its
    # first word past the declared positionals.
    $ended = $false
    $swallow = $false
    $tail = $false
    $free = 0
    $isVerbatim = $clawwrightVerbatim.ContainsKey($cmd)
    foreach ($w2 in $between) {
      if ($swallow) { $swallow = $false; continue }
      if ($w2 -ceq '--') { $ended = $true; break }
      $key = $w2
      $inline = $false
      if ($w2.StartsWith('--') -and $w2.Contains('=')) { $key = $w2.Substring(0, $w2.IndexOf('=')); $inline = $true }
      if ((-not $inline) -and ($w2 -cne '') -and ($clawwrightValueOptions[$cmd] -ccontains $w2)) { $swallow = $true; continue }
      if ($w2.StartsWith('-')) {
        if ($isVerbatim -and -not ($clawwrightVerbatimFlags[$cmd] -ccontains $key)) { $tail = $true; break }
        continue
      }
      $free = $free + 1
      if ($isVerbatim -and $free -gt $clawwrightVerbatim[$cmd]) { $tail = $true; break }
    }
    # The first word names the action; an option still waiting for its value completes that
    # value (keyed command + action + option), or nothing when it has no choices.
    $scope = ''
    $afterKey = "$cmd $($between[0])"
    if ($between.Count -gt 0 -and $clawwrightAfter.ContainsKey($afterKey)) { $scope = $between[0] }
    if ($between.Count -eq 0) {
      # An unknown command reaches `first` with nothing under that key: the @() keeps the
      # type an array and the filter below drops the $null — nothing is offered.
      $candidates = @($clawwrightFirst[$cmd])
    } elseif ($ended -or $tail) {
      $candidates = @()
    } elseif ($swallow) {
      $key = "$cmd$scope$prev"
      if ($clawwrightValues.ContainsKey($key)) { $candidates = @($clawwrightValues[$key]) } else { $candidates = @() }
    } elseif ($clawwrightAfter.ContainsKey($afterKey)) {
      $candidates = @($clawwrightAfter[$afterKey])
    } else {
      $candidates = @($clawwrightAfter["$cmd *"])
    }
  }
  $candidates | Where-Object { $_ -clike "$wordToComplete*" } | Sort-Object -Unique -CaseSensitive | ForEach-Object {
    [System.Management.Automation.CompletionResult]::new($_, $_, 'ParameterValue', $_)
  }
}
Register-ArgumentCompleter -Native -CommandName clawwright, ./clawwright -ScriptBlock $clawwrightCompleter
